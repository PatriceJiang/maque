# maque::platform -- native SDL2 window/input, or a headless WASM P0 stub.
if(MAQUE_WASM OR EMSCRIPTEN OR MAQUE_HEADLESS_STUBS)
    add_library(maque_platform STATIC src/platform_stub.cpp)
    add_library(maque::platform ALIAS maque_platform)
    target_include_directories(maque_platform PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/include"
        "${MAQUE_GEN_DIR}")
    target_link_libraries(maque_platform PUBLIC maque::tracy)
    set_target_properties(maque_platform PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON)
    message(STATUS "maque::platform: headless stub enabled")
else()
    if(TARGET SDL2::SDL2)
        set(_maque_sdl2 SDL2::SDL2)
    elseif(TARGET SDL2)
        set(_maque_sdl2 SDL2)
    else()
        find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2)
        find_library(SDL2_LIBRARY NAMES SDL2)
        if(SDL2_INCLUDE_DIR AND SDL2_LIBRARY)
            set(_maque_sdl2 "${SDL2_LIBRARY}")
        endif()
    endif()

    if(_maque_sdl2)
        add_library(maque_platform STATIC src/platform.cpp)
        add_library(maque::platform ALIAS maque_platform)
        target_include_directories(maque_platform PUBLIC
            "${CMAKE_CURRENT_SOURCE_DIR}/include"
            "${MAQUE_GEN_DIR}")
        if(SDL2_INCLUDE_DIR)
            target_include_directories(maque_platform PUBLIC "${SDL2_INCLUDE_DIR}")
        endif()
        target_link_libraries(maque_platform PUBLIC ${_maque_sdl2} maque::tracy)
        set_target_properties(maque_platform PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON)
        if(WIN32)
            target_link_libraries(maque_platform PUBLIC psapi)
        endif()

        if(MAQUE_BUILD_TESTS)
            add_executable(maque_platform_test tests/platform_test.cpp)
            maque_deploy_asan(maque_platform_test)
            target_link_libraries(maque_platform_test PRIVATE maque::platform)
            add_test(NAME platform COMMAND maque_platform_test)
        endif()
    else()
        message(WARNING "maque: SDL2 not found; platform module disabled")
    endif()
endif()
